  @override
  void initState() {

  void _onSearchChanged() {
    _filterConversations();
  }
  
  void _filterConversations() {
    final query = _searchController.text.toLowerCase();
    
    setState(() {
      if (query.isEmpty) {
        // Apply only category filter
        _applyFilter(_activeFilter);
      } else {
        // Apply both search and category filter
        _filteredConversations = _allConversations.where((conversation) {
          final matchesSearch = conversation["name"].toString().toLowerCase().contains(query);
          final matchesFilter = _matchesFilter(conversation, _activeFilter);
          return matchesSearch && matchesFilter;
        }).toList();
      }
    });
  }
  
  bool _matchesFilter(Map<String, dynamic> conversation, String filter) {
    if (filter == "All") return true;
    if (filter == "Unread") return conversation["unread"] == true;
    if (filter == "Favorites") return conversation["isFavorite"] == true;
    return true;
  }
  
  void _applyFilter(String filter) {
    setState(() {
      _activeFilter = filter;
      
      if (_searchController.text.isNotEmpty) {
        _filterConversations();
        return;
      }
      
      if (filter == "All") {
        _filteredConversations = List.from(_allConversations);
      } else if (filter == "Unread") {
        _filteredConversations = _allConversations.where((conversation) => 
          conversation["unread"] == true
        ).toList();
      } else if (filter == "Favorites") {
        _filteredConversations = _allConversations.where((conversation) => 
          conversation["isFavorite"] == true
        ).toList();
      }
    });
  }
  
  void _toggleFavorite(String id) {
    setState(() {
      // Update in the original list
      for (var conversation in _allConversations) {
        if (conversation["id"] == id) {
          conversation["isFavorite"] = !conversation["isFavorite"];
          break;
        }
      }
      
      // Update in the filtered list
      for (var conversation in _filteredConversations) {
        if (conversation["id"] == id) {
          conversation["isFavorite"] = !conversation["isFavorite"];
          break;
        }
      }
    });
  }
  
  void _markAsRead(String id) {
    setState(() {
      // Update in the original list
      for (var conversation in _allConversations) {
        if (conversation["id"] == id) {
          conversation["unread"] = false;
          break;
        }
      }
      
      // Update in the filtered list
      for (var conversation in _filteredConversations) {
        if (conversation["id"] == id) {
          conversation["unread"] = false;
          break;
        }
      }
    });
  }
  
  void _deleteConversation(String id) {
    setState(() {
      // Remove from the original list
      _allConversations.removeWhere((conversation) => conversation["id"] == id);
      
      // Remove from the filtered list
      _filteredConversations.removeWhere((conversation) => conversation["id"] == id);
    });
    
    // Show confirmation
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: const Text("Conversation deleted"),
        backgroundColor: Colors.grey.shade800,
        action: SnackBarAction(
          label: "UNDO",
          onPressed: () {
            // Implement undo functionality
          },
          textColor: Colors.white,
        ),
      ),
    );
  }

  void _navigateToDiscover(BuildContext context) {
    Navigator.pushReplacement(
      context,
      PageRouteBuilder(
        pageBuilder: (context, animation, secondaryAnimation) =>
            const DiscoverScreen(),
        transitionsBuilder: (context, animation, secondaryAnimation, child) {
          const begin = Offset(-1.0, 0.0);
          const end = Offset.zero;
          const curve = Curves.easeInOut;

          var tween =
              Tween(begin: begin, end: end).chain(CurveTween(curve: curve));
          var offsetAnimation = animation.drive(tween);

          return SlideTransition(
            position: offsetAnimation,
            child: child,
          );
        },
        transitionDuration: const Duration(milliseconds: 300),
      ),
    );
  }

        return _buildConversationItem(context, conversation);
      },
    );
  }

  Widget _buildConversationItem(BuildContext context, Map<String, dynamic> conversation) {
    return Dismissible(
      key: Key(conversation["id"]),
      background: Container(
        alignment: Alignment.centerLeft,
        padding: const EdgeInsets.only(left: 20),
        color: Colors.blue,
        child: const Icon(Icons.check_circle, color: Colors.white),
      ),
      secondaryBackground: Container(
        alignment: Alignment.centerRight,
        padding: const EdgeInsets.only(right: 20),
        color: Colors.red,
        child: const Icon(Icons.delete, color: Colors.white),
      ),
      confirmDismiss: (direction) async {
        if (direction == DismissDirection.startToEnd) {
          // Mark as read
          _markAsRead(conversation["id"]);
          return false; // Don't dismiss the item
        } else {
          // Delete conversation
          return await showDialog(
            context: context,
            builder: (BuildContext context) {
              return AlertDialog(
                title: const Text("Delete Conversation"),
                content: Text("Are you sure you want to delete your conversation with ${conversation["name"]}?"),
                actions: [
                  TextButton(
                    onPressed: () => Navigator.of(context).pop(false),
                    child: const Text("CANCEL"),
                  ),
                  TextButton(
                    onPressed: () => Navigator.of(context).pop(true),
                    child: const Text("DELETE"),
                  ),
                ],
              );
            },
          );
        }
      },
      onDismissed: (direction) {
        if (direction == DismissDirection.endToStart) {
          _deleteConversation(conversation["id"]);
        }
      },
      child: AnimatedPressable(
        onPressed: () => _navigateToConversation(context, conversation),
        child: Container(
          padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 12),
          margin: const EdgeInsets.symmetric(horizontal: 16, vertical: 4),
          decoration: BoxDecoration(
            color: Colors.white,
            borderRadius: BorderRadius.circular(12),
            boxShadow: [
              BoxShadow(
                color: Colors.black.withOpacity(0.05),
                blurRadius: 2,
                offset: const Offset(0, 1),
              ),
            ],
          ),
          child: Row(
            children: [
              // Profile image with status indicator
              Stack(
                children: [
                  // Profile image
                  Container(
                    width: 60,
                    height: 60,
                    decoration: BoxDecoration(
                      shape: BoxShape.circle,
                      border: Border.all(
                        color: conversation["isSystem"] == true
                            ? Colors.purple
                            : AppColors.primaryColor,
                        width: 2,
                      ),
                    ),
                    child: ClipRRect(
                      borderRadius: BorderRadius.circular(30),
                      child: conversation["isSystem"] == true
                          ? Container(
                              color: Colors.purple.shade100,
                              child: const Center(
                                child: Text(
                                  "A",
                                  style: TextStyle(
                                    color: Colors.purple,
                                    fontWeight: FontWeight.bold,
                                    fontSize: 20,
                                  ),
                                ),
                              ),
                            )
                          : Image.asset(
                              conversation["image"],
                              fit: BoxFit.cover,
                            ),
                    ),
                  ),
                  
                  // Online indicator
                  if (conversation["isOnline"] == true)
                    Positioned(
                      right: 0,
                      bottom: 0,
                      child: Container(
                        width: 16,
                        height: 16,
                        decoration: BoxDecoration(
                          color: Colors.green,
                          shape: BoxShape.circle,
                          border: Border.all(
                            color: Colors.white,
                            width: 2,
                          ),
                        ),
                      ),
                    ),
                ],
              ),
              const SizedBox(width: 12),

              // Message preview
              Expanded(
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    // Name and favorite indicator
                    Row(
                      children: [
                        Text(
                          conversation["name"],
                          style: TextStyle(
                            fontWeight: conversation["unread"] == true
                                ? FontWeight.bold
                                : FontWeight.normal,
                            fontSize: 16,
                          ),
                        ),
                        const SizedBox(width: 4),
                        if (conversation["isFavorite"] == true)

                            Icons.star,

                            size: 14,

                      ],
                    ),
                    const SizedBox(height: 4),
                    
                    // Message or typing indicator
                    conversation["isTyping"] == true
                        ? Row(
                            children: [
                              Text(
                                "typing",
                                style: TextStyle(
                                  color: AppColors.primaryColor,
                                  fontStyle: FontStyle.italic,
                                  fontSize: 14,
                                ),
                              ),
                              const SizedBox(width: 4),
                              _buildTypingIndicator(),
                            ],
                          )
                        : Text(
                            conversation["message"],
                            style: TextStyle(
                              color: conversation["unread"] == true
                                  ? Colors.black87
                                  : Colors.grey.shade600,
                              fontWeight: conversation["unread"] == true
                                  ? FontWeight.w500
                                  : FontWeight.normal,
                              fontSize: 14,
                            ),
                            maxLines: 1,
                            overflow: TextOverflow.ellipsis,
                          ),
                  ],
                ),
              ),

              // Time and unread indicator
              Column(
                crossAxisAlignment: CrossAxisAlignment.end,
                children: [
                  Text(
                    conversation["time"],
                    style: TextStyle(
                      color: conversation["unread"] == true
                          ? AppColors.primaryColor
                          : Colors.grey.shade600,
                      fontWeight: conversation["unread"] == true
                          ? FontWeight.bold
                          : FontWeight.normal,
                      fontSize: 12,
                    ),
                  ),
                  const SizedBox(height: 4),
                  if (conversation["unread"] == true)
                    Container(
                      padding: const EdgeInsets.symmetric(horizontal: 6, vertical: 2),
                      decoration: BoxDecoration(
                        color: AppColors.primaryColor,
                        borderRadius: BorderRadius.circular(10),
                      ),
                      child: const Text(
                        "NEW",
                        style: TextStyle(
                          color: Colors.white,
                          fontSize: 10,
                          fontWeight: FontWeight.bold,
                        ),
                      ),
                    ),
                ],
              ),
            ],
          ),
        ),
      ),
    );
  }
  
  Widget _buildTypingIndicator() {
    return SizedBox(
      width: 30,
      height: 15,
      child: Row(
        mainAxisAlignment: MainAxisAlignment.spaceEvenly,
        children: List.generate(3, (index) {
          return _buildPulsingDot(
            delay: Duration(milliseconds: 300 * index),
          );
        }),
      ),
    );
  }
  
  Widget _buildPulsingDot({required Duration delay}) {
    return TweenAnimationBuilder<double>(
      tween: Tween(begin: 0.0, end: 1.0),
      duration: const Duration(milliseconds: 1000),
      curve: Curves.easeInOut,
      builder: (context, value, child) {
        return Container(
          width: 5 * (0.5 + (value * 0.5)),
          height: 5 * (0.5 + (value * 0.5)),
          decoration: BoxDecoration(
            shape: BoxShape.circle,
            color: AppColors.primaryColor.withOpacity(0.5 + (value * 0.5)),
          ),
        );
      },
    );
  }

          children: [
            Icon(
              icon,

            if (hasBadge)
              Positioned(
                right: -2,
                top: -2,
                child: Container(
                  width: 10,
                  height: 10,
                  decoration: BoxDecoration(
                    color: AppColors.primaryColor,
                    shape: BoxShape.circle,
                  ),
                ),
              ),
          ],
        ),
        const SizedBox(height: 4),
        Text(
          label,
          style: TextStyle(
            color: isActive ? AppColors.primaryColor : Colors.grey,
            fontSize: 12,
          ),
        ),
      ],
    );
  }
