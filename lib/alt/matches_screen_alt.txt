  void _applyFilter(String filter) {
    setState(() {
      _activeFilter = filter;
      
      if (filter == "All") {
        _filteredMatches = List.from(_allMatches);
      } else if (filter == "New") {
        _filteredMatches = _allMatches.where((match) => match["isNew"] == true).toList();
      } else if (filter == "Favorites") {
        _filteredMatches = _allMatches.where((match) => match["isFavorite"] == true).toList();
      } else if (filter == "Online") {
        _filteredMatches = _allMatches.where((match) => match["isOnline"] == true).toList();
      }
    });
  }
  
  void _toggleFavorite(int index) {
    setState(() {
      final matchId = _filteredMatches[index]["name"];
      
      // Find and update in the original list
      for (int i = 0; i < _allMatches.length; i++) {
        if (_allMatches[i]["name"] == matchId) {
          _allMatches[i]["isFavorite"] = !_allMatches[i]["isFavorite"];
          break;
        }
      }
      
      // Update in the filtered list
      _filteredMatches[index]["isFavorite"] = !_filteredMatches[index]["isFavorite"];
    });
  }

  // Add method to navigate back to Discover with animation
  void _navigateToDiscover(BuildContext context) {
    Navigator.pushReplacement(
      context,
      PageRouteBuilder(
        pageBuilder: (context, animation, secondaryAnimation) =>
            const DiscoverScreen(),
        transitionsBuilder: (context, animation, secondaryAnimation, child) {
          const begin =
              Offset(-1.0, 0.0); // Start from left (reverse direction)
          const end = Offset.zero;
          const curve = Curves.easeInOut;

          var tween =
              Tween(begin: begin, end: end).chain(CurveTween(curve: curve));
          var offsetAnimation = animation.drive(tween);

          return SlideTransition(
            position: offsetAnimation,
            child: child,
          );
        },
        transitionDuration: const Duration(milliseconds: 300),
      ),
    );
  }

      itemBuilder: (context, index) {

  Widget _buildBottomNavBar(BuildContext context) {
    return Container(
      height: 60,
      decoration: BoxDecoration(
        color: Colors.white,
        boxShadow: [
          BoxShadow(
            color: Colors.black.withOpacity(0.1),

      ),
      child: Row(
        mainAxisAlignment: MainAxisAlignment.spaceAround,

            onTap: () => _navigateToDiscover(context),
            child: _buildNavItem(Icons.search, "Discover", false),
          ),
          _buildNavItem(Icons.favorite, "Matches", true),
          GestureDetector(
            onTap: () => _navigateToMessages(context),
            child: _buildNavItem(Icons.chat_bubble_outline, "Messages", false,
                hasBadge: true),
          ),
          GestureDetector(
            onTap: () => _navigateToProfile(context),
            child: _buildNavItem(Icons.person_outline, "Profile", false),
          ),
        ],
      ),
    );

                    color: AppColors.primaryColor,

              ),

        Text(
          label,
          style: TextStyle(

    );
