import "package:flutter/material.dart";
import "dart:async";
import "../../utils/constants.dart";
import "../../utils/animations.dart";
import "package:google_fonts/google_fonts.dart";

class ConversationScreen extends StatefulWidget {
  final Map<String, dynamic> conversation;

  const ConversationScreen({
    super.key,
    required this.conversation,
  });

  @override
  State<ConversationScreen> createState() => _ConversationScreenState();
}

class _ConversationScreenState extends State<ConversationScreen>
    with SingleTickerProviderStateMixin {
  final TextEditingController _messageController = TextEditingController();
  final ScrollController _scrollController = ScrollController();
  final List<Map<String, dynamic>> _messages = [];
  bool _showAttachmentMenu = false;
  bool _showEmojiPicker = false;
  late AnimationController _animationController;
  late Animation<double> _animation;
  bool _showHeartAnimation = false;
  Timer? _heartAnimationTimer;
  bool _showCompactHeader = false;
  bool _isRecording = false;
  bool _isTyping = false;
  Timer? _typingTimer;
  bool _showScrollToBottom = false;
  int _unreadCount = 0;
  
  // Selected message for reply
  Map<String, dynamic>? _replyToMessage;
  
  // Media gallery
  final List<String> _sharedMedia = [
    "assets/images/shared_photo1.jpg",
    "assets/images/shared_photo2.jpg",
    "assets/images/shared_photo3.jpg",
  ];

  @override
  void initState() {
    super.initState();

    // Initialize animation controller for attachment menu
    _animationController = AnimationController(
      vsync: this,
      duration: const Duration(milliseconds: 200),
    );

    _animation = CurvedAnimation(
      parent: _animationController,
      curve: Curves.easeOut,
    );

    // Initialize messages from conversation
    _messages.addAll(
        List<Map<String, dynamic>>.from(widget.conversation["messages"] ?? []));

    // Add listener to scroll controller to detect when to show compact header
    _scrollController.addListener(_onScroll);
    
    // Simulate typing after a delay
    Future.delayed(const Duration(seconds: 3), () {
      if (mounted) {
        _simulateTyping();
      }
    });
  }

  void _onScroll() {
    if (_scrollController.hasClients) {
      if (_scrollController.offset > 150 && !_showCompactHeader) {
        setState(() {
          _showCompactHeader = true;
        });
      } else if (_scrollController.offset <= 150 && _showCompactHeader) {
        setState(() {
          _showCompactHeader = false;
        });
      }
      
      // Show scroll to bottom button when scrolled up
      if (_scrollController.position.maxScrollExtent - _scrollController.offset > 300) {
        if (!_showScrollToBottom) {
          setState(() {
            _showScrollToBottom = true;
          });
        }
      } else {
        if (_showScrollToBottom) {
          setState(() {
            _showScrollToBottom = false;
          });
        }
      }
    }
  }
  
  void _simulateTyping() {
    setState(() {
      _isTyping = true;
    });
    
    // After 3 seconds, send a message
    Future.delayed(const Duration(seconds: 3), () {
      if (mounted) {
        setState(() {
          _isTyping = false;
          
          // Add automated message
          _messages.add({
            "sender": "other",
            "text": "I'm really enjoying our conversation! What are your plans for the weekend?",
            "time": _getCurrentTime(),
            "status": "sent",
          });
          
          // Increment unread count if not at bottom
          if (_scrollController.hasClients && 
              _scrollController.position.maxScrollExtent - _scrollController.offset > 100) {
            _unreadCount++;
          } else {
            // Scroll to bottom
            _scrollToBottom();
          }
        });
      }
    });
  }

  @override
  void dispose() {
    _messageController.dispose();
    _scrollController.dispose();
    _animationController.dispose();
    _heartAnimationTimer?.cancel();
    _typingTimer?.cancel();
    super.dispose();
  }

  void _navigateBack(BuildContext context) {
    Navigator.pop(context);
  }

  void _toggleAttachmentMenu() {
    setState(() {
      _showAttachmentMenu = !_showAttachmentMenu;
      if (_showEmojiPicker) {
        _showEmojiPicker = false;
      }
      if (_showAttachmentMenu) {
        _animationController.forward();
      } else {
        _animationController.reverse();
      }
    });
  }
  
  void _toggleEmojiPicker() {
    setState(() {
      _showEmojiPicker = !_showEmojiPicker;
      if (_showAttachmentMenu) {
        _showAttachmentMenu = false;
        _animationController.reverse();
      }
    });
  }

  void _showFeatureComingSoonMessage() {
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: const Text("This feature will be available soon!"),
        backgroundColor: AppColors.primaryColor,
        behavior: SnackBarBehavior.floating,
        shape: RoundedRectangleBorder(
          borderRadius: BorderRadius.circular(10),
        ),
        margin: const EdgeInsets.all(10),
      ),
    );

    // Close the attachment menu
    if (_showAttachmentMenu) {
      _toggleAttachmentMenu();
    }
  }
  
  void _startRecording() {
    setState(() {
      _isRecording = true;
    });
    
    // Show a message
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: const Text("Recording voice message..."),
        backgroundColor: Colors.green,
        duration: const Duration(seconds: 60),
        action: SnackBarAction(
          label: "CANCEL",
          textColor: Colors.white,
          onPressed: _cancelRecording,
        ),
      ),
    );
  }
  
  void _cancelRecording() {
    setState(() {
      _isRecording = false;
    });
    
    ScaffoldMessenger.of(context).hideCurrentSnackBar();
  }
  
  void _sendVoiceMessage() {
    setState(() {
      _isRecording = false;
    });
    
    ScaffoldMessenger.of(context).hideCurrentSnackBar();
    
    // Add voice message
    setState(() {
      _messages.add({
        "sender": "me",
        "isVoice": true,
        "duration": "0:12",
        "time": _getCurrentTime(),
        "status": "sent",
      });
    });
    
    // Scroll to bottom
    _scrollToBottom();
    
    // Simulate reply after a delay
    Future.delayed(const Duration(seconds: 2), () {
      if (mounted) {
        _simulateTyping();
      }
    });
  }
  
  void _setReplyMessage(Map<String, dynamic> message) {
    setState(() {
      _replyToMessage = message;
    });
    
    // Focus on text field
    FocusScope.of(context).requestFocus(FocusNode());
    Future.delayed(const Duration(milliseconds: 100), () {
      FocusScope.of(context).requestFocus(FocusNode());
    });
  }
  
  void _cancelReply() {
    setState(() {
      _replyToMessage = null;
    });
  }
  
  void _showMediaGallery() {
    showModalBottomSheet(
      context: context,
      backgroundColor: Colors.black,
      builder: (context) => _buildMediaGallery(),
    );
  }

  void _sendMessage(String text) {
    if (text.trim().isEmpty && !_isRecording) return;
    
    if (_isRecording) {
      _sendVoiceMessage();
      return;
    }

    setState(() {
      // Add user message
      _messages.add({
        "sender": "me",
        "text": text,
        "time": _getCurrentTime(),
        "status": "sent",
        "replyTo": _replyToMessage,
      });

      // Clear input field and reply
      _messageController.clear();
      _replyToMessage = null;
      
      // Reset unread count
      _unreadCount = 0;
    });

    // Scroll to bottom
    _scrollToBottom();
    
    // Start typing indicator after a delay
    _typingTimer?.cancel();
    _typingTimer = Timer(const Duration(seconds: 2), () {
      if (mounted) {
        _simulateTyping();
      }
    });
  }

  void _sendHeart() {
    setState(() {
      // Add heart message
      _messages.add({
        "sender": "me",
        "isHeart": true,
        "time": _getCurrentTime(),
        "status": "sent",
      });

      // Show heart animation
      _showHeartAnimation = true;
      
      // Reset unread count
      _unreadCount = 0;
    });

    // Scroll to bottom
    _scrollToBottom();

    // Hide heart animation after a delay
    _heartAnimationTimer?.cancel();
    _heartAnimationTimer = Timer(const Duration(milliseconds: 1500), () {
      if (mounted) {
        setState(() {
          _showHeartAnimation = false;
        });
      }
    });

    // Add automated reply after a short delay
    Timer(const Duration(milliseconds: 1000), () {
      if (mounted) {
        setState(() {
          _messages.add({
            "sender": "other",
            "isHeart": true,
            "time": _getCurrentTime(),
            "status": "sent",
          });
        });

        // Scroll to bottom again
        _scrollToBottom();
      }
    });
  }
  
  void _sendImage() {
    // Close attachment menu
    if (_showAttachmentMenu) {
      _toggleAttachmentMenu();
    }
    
    setState(() {
      // Add image message
      _messages.add({
        "sender": "me",
        "isImage": true,
        "imageUrl": "assets/images/shared_photo1.jpg",
        "time": _getCurrentTime(),
        "status": "sent",
      });
      
      // Reset unread count
      _unreadCount = 0;
    });
    
    // Scroll to bottom
    _scrollToBottom();
    
    // Start typing indicator after a delay
    _typingTimer?.cancel();
    _typingTimer = Timer(const Duration(seconds: 3), () {
      if (mounted) {
        _simulateTyping();
      }
    });
  }

  String _getCurrentTime() {
    final now = DateTime.now();
    final hour = now.hour > 12 ? now.hour - 12 : now.hour == 0 ? 12 : now.hour;
    final period = now.hour >= 12 ? "PM" : "AM";
    final minute = now.minute.toString().padLeft(2, "0");
    return "$hour:$minute $period";
  }

  void _scrollToBottom() {
    if (_scrollController.hasClients) {
      Timer(const Duration(milliseconds: 100), () {
        _scrollController.animateTo(
          _scrollController.position.maxScrollExtent,
          duration: const Duration(milliseconds: 300),
          curve: Curves.easeOut,
        );
      });
    }
    
    // Reset unread count
    if (_unreadCount > 0) {
      setState(() {
        _unreadCount = 0;
      });
    }
  }

  Widget _buildPulsingDot({required Duration delay}) {
    return TweenAnimationBuilder<double>(
      tween: Tween(begin: 0.0, end: 1.0),
      duration: const Duration(milliseconds: 1000),
      curve: Curves.easeInOut,
      builder: (context, value, child) {
        return Container(
          width: 5 * (0.5 + (value * 0.5)),
          height: 5 * (0.5 + (value * 0.5)),

  Widget _buildMessagesList() {
    return Container(
      decoration: BoxDecoration(
        color: Colors.grey.shade100,
        image: const DecorationImage(
          image: AssetImage("assets/images/chat_bg.png"),
          fit: BoxFit.cover,
          opacity: 0.1,
        ),
      ),
      child: ListView.builder(
        controller: _scrollController,
        padding: const EdgeInsets.all(16),
        itemCount: _messages.length + (_isTyping ? 1 : 0),
        itemBuilder: (context, index) {
          if (index == _messages.length && _isTyping) {
            return _buildTypingMessage();
          }
          
          final message = _messages[index];
          final isMe = message["sender"] == "me";
          
          if (message["isHeart"] == true) {
            return _buildHeartMessage(isMe, message["time"]);
          } else if (message["isVoice"] == true) {
            return _buildVoiceMessage(isMe, message);
          } else if (message["isImage"] == true) {
            return _buildImageMessage(isMe, message);
          } else {
            return _buildTextMessage(isMe, message);
          }
        },
      ),
    );
  }

class WaveformPainter extends CustomPainter {
  final Color color;
  final int waveCount;
  
  WaveformPainter({
    required this.color,
    required this.waveCount,
  });
  
  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = color
      ..strokeWidth = 2
      ..strokeCap = StrokeCap.round;
    
    final width = size.width;
    final height = size.height;
    final barWidth = width / (waveCount * 2);
    
    for (int i = 0; i < waveCount; i++) {
      final x = i * barWidth * 2 + barWidth;
      final barHeight = (i % 3 == 0) 
          ? height * 0.7 
          : (i % 2 == 0) 
              ? height * 0.5 
              : height * 0.3;
      
      final startY = (height - barHeight) / 2;
      final endY = startY + barHeight;
      
      canvas.drawLine(
        Offset(x, startY),
        Offset(x, endY),
        paint,
      );
    }
  }
  
  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) {
    return false;
  }
}
